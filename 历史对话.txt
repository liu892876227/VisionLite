              HTuple startCol = _interactiveCaliper.GetDrawingObjectParams("column1");
              HTuple endRow = _interactiveCaliper.GetDrawingObjectParams("row2");
              HTuple endCol = _interactiveCaliper.GetDrawingObjectParams("column2");

              AlgorithmParameterPanel.UpdateParameterValueSilently("StartRow", startRow.D);
              AlgorithmParameterPanel.UpdateParameterValueSilently("StartCol", startCol.D);
              AlgorithmParameterPanel.UpdateParameterValueSilently("EndRow", endRow.D);
              AlgorithmParameterPanel.UpdateParameterValueSilently("EndCol", endCol.D);

              // 更新处理器参数
              if (_currentProcessor is LineCaliperProcessor lineProcessor)
              {
                  lineProcessor.StartRow = startRow.D;
                  lineProcessor.StartCol = startCol.D;
                  lineProcessor.EndRow = endRow.D;
                  lineProcessor.EndCol = endCol.D;
              }

              UpdateStatus($"直线ROI: ({startRow.D:F0},{startCol.D:F0}) → ({endRow.D:F0},{endCol.D:F0})");

              // 触发实时预览
              if (_originalImage != null)
              {
                  TriggerRealtimePreview("ROI");
              }
          }
          finally
          {
              _isUpdatingFromROI = false;
          }
      });
  }

  3. 卡尺分布策略设计

  3.1 直线上卡尺位置计算

  /// <summary>
  /// 计算直线上的卡尺位置
  /// 在直线上均匀分布指定数量的卡尺
  /// </summary>
  /// <param name="startRow">起点行坐标</param>
  /// <param name="startCol">起点列坐标</param>
  /// <param name="endRow">终点行坐标</param>
  /// <param name="endCol">终点列坐标</param>
  /// <returns>卡尺信息列表</returns>
  private List<CaliperInfo> CalculateCaliperPositions(double startRow, double startCol, double endRow, double endCol)
  {
      var calipers = new List<CaliperInfo>();

      try
      {
          // 计算直线长度和方向
          double lineLength = Math.Sqrt(Math.Pow(endRow - startRow, 2) + Math.Pow(endCol - startCol, 2));
          double lineAngle = Math.Atan2(endCol - startCol, endRow - startRow);

          // 计算步长
          double stepLength = lineLength / (CaliperCount + 1);

          for (int i = 0; i < CaliperCount; i++)
          {
              // 计算当前卡尺在直线上的位置（从起点开始的距离）
              double distance = (i + 1) * stepLength;

              // 计算卡尺中心位置
              double caliperRow = startRow + distance * Math.Cos(lineAngle);
              double caliperCol = startCol + distance * Math.Sin(lineAngle);

              // 卡尺方向：垂直于直线方向
              double caliperPhi = lineAngle + Math.PI / 2.0;

              // 创建卡尺信息
              var caliper = new CaliperInfo(
                  caliperRow,
                  caliperCol,
                  caliperPhi,
                  CaliperLength,
                  CaliperWidth
              );

              calipers.Add(caliper);
          }

          return calipers;
      }
      catch (Exception ex)
      {
          throw new InvalidOperationException($"计算卡尺位置失败: {ex.Message}", ex);
      }
  }

  4.2 直线拟合实现

  /// <summary>
  /// 从边缘点拟合直线
  /// 使用Halcon的fit_line_contour_xld算子
  /// </summary>
  /// <param name="edgePoints">边缘点列表</param>
  /// <returns>直线拟合结果</returns>
  private LineFitResult FitLineFromEdgePoints(List<EdgePoint> edgePoints)
  {
      var result = new LineFitResult();

      try
      {
          if (edgePoints.Count < MinFitPoints)
          {
              result.ErrorMessage = $"边缘点数量不足，需要至少 {MinFitPoints} 个点，实际只有 {edgePoints.Count} 个";
              return result;
          }

          // 转换为HTuple数组
          var rowArray = edgePoints.Select(p => p.Row).ToArray();
          var colArray = edgePoints.Select(p => p.Col).ToArray();
          var rowTuple = new HTuple(rowArray);
          var colTuple = new HTuple(colArray);

          // 生成轮廓
          HOperatorSet.GenContourPolygonXld(out HObject contour, rowTuple, colTuple);

          // 拟合直线
          HOperatorSet.FitLineContourXld(
              contour,                        // 输入轮廓
              GetLineFittingAlgorithmString(), // 拟合算法
              -1,                             // 最大迭代次数（-1表示自动）
              0,                              // 随机采样点数（0表示使用所有点）
              0,                              // 随机种子
              MinFitPoints,                   // 最小点数
              ClippingFactor,                 // 异常点抑制因子
              out HTuple rowBegin,            // 直线起点行坐标
              out HTuple colBegin,            // 直线起点列坐标
              out HTuple rowEnd,              // 直线终点行坐标
              out HTuple colEnd,              // 直线终点列坐标
              out HTuple nr,                  // 直线法向量行分量
              out HTuple nc,                  // 直线法向量列分量
              out HTuple dist);               // 点到直线的距离

          // 计算拟合误差
          double fitError = CalculateLineFitError(edgePoints, rowBegin.D, colBegin.D, rowEnd.D, colEnd.D);

          // 填充结果
          result.IsValid = true;
          result.StartRow = rowBegin.D;
          result.StartCol = colBegin.D;
          result.EndRow = rowEnd.D;
          result.EndCol = colEnd.D;
          result.FitError = fitError;
          result.ValidEdgeCount = edgePoints.Count;

          contour?.Dispose();
          return result;
      }
      catch (Exception ex)
      {
          result.ErrorMessage = $"直线拟合失败: {ex.Message}";
          return result;
      }
  }
